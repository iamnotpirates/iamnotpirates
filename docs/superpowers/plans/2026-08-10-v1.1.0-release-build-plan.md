```python
"""Automated release build script for IAmNotPirates.exe v1.1.0."""
import os
import shutil
import subprocess
import sys
from pathlib import Path

def clean_build_artifacts():
    for folder in ["build", "dist"]:
        if os.path.exists(folder):
            shutil.rmtree(folder, ignore_errors=True)

def build_executable():
    print("[Build] Cleaning build artifacts...")
    clean_build_artifacts()
    
    print("[Build] Compiling IAmNotPirates.exe via PyInstaller...")
    cmd = [sys.executable, "-m", "PyInstaller", "IAmNotPirates.spec", "--noconfirm"]
    res = subprocess.run(cmd, check=False)
    if res.returncode != 0:
        print("[Build] PyInstaller compilation failed.")
        sys.exit(1)
        
    exe_path = Path("dist/IAmNotPirates.exe")
    if exe_path.exists():
        size_mb = exe_path.stat().st_size / (1024 * 1024)
        print(f"[Build] SUCCESS! Executable built at: {exe_path.resolve()} ({size_mb:.2f} MB)")
    else:
        print("[Build] Output executable dist/IAmNotPirates.exe not found.")
        sys.exit(1)

if __name__ == "__main__":
    build_executable()
```
